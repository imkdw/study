## SmartDiskWriter

```ts
import { close, open, write } from "node:fs";
import { Writable } from "node:stream";

interface SmartDiskWriterOptions {
  highWaterMark: number;
  destination: string;
}

type StreamCallback = (error?: Error | null) => void;

/**
 * Writable은 코어 레벨에서 미리 정의해둔 베이스 객체임
 * 복잡한 이벤트 제어 구현을 생략할 수 있고 백프레셔나 에러처리 등 모두 제공받음
 */
export class SmartDiskWriter extends Writable {
  private readonly destination: string;
  private fileDescriptor: number | null = null;
  private ramBuffer: Buffer[] = [];
  private bufferedBytes = 0;
  private ioCount = 0;

  constructor({ highWaterMark, destination }: SmartDiskWriterOptions) {
    /**
     * 백프레셔의 기준점을 설정함
     */
    super({ highWaterMark });

    this.destination = destination;
  }

  /**
   * 새로 만든 클래스를 호출한 직후에 다른 데이터 쓰기작업 시작 전 Node.js에서 자동 실행되는 첫 단계
   * 직접 해당 메서드를 실행하지 않아도 내부적인 생명 주기에 맞춰서 알아서 호출해줌
   */
  _construct(callback: StreamCallback): void {
    /**
     * 시스템에게 파일 쓰기작업을 위한 권한을 요청함
     * 이후 커널에서는 파일 디스크립터를 넘겨주며 이를 보관해놓음
     */
    open(this.destination, "w", (error, fd) => {
      if (error) {
        callback(error);
      } else {
        this.fileDescriptor = fd;
        callback();
      }
    });
  }

  /**
   * 파이프에 데이터를 밀어넣을 때 마다 반복해서 호출되는 핵심 기능
   */
  _write(chunk: Buffer, _encoding: BufferEncoding, callback: StreamCallback): void {
    this.ramBuffer.push(chunk);
    this.bufferedBytes += chunk.length;

    /**
     * 디스크의 속도 보완을 위해서 최대 보관하는 용량이 꽉 차면 그 때 디스크에 데이터를 기록함
     */
    if (this.bufferedBytes >= this.writableHighWaterMark) {
      /**
       * _construct 에서 열어둔 fd 가 없으면 쓸 대상이 없으므로 여기서 걸러냄
       * 타입 관점에서도 number | null 을 number 로 좁혀준다
       */
      if (this.fileDescriptor === null) {
        return callback(new Error("파일 디스크립터가 아직 준비되지 않았습니다"));
      }

      write(this.fileDescriptor, Buffer.concat(this.ramBuffer), (error) => {
        if (error) {
          return callback(error);
        }

        this.ramBuffer = [];
        this.bufferedBytes = 0;

        /**
         * this.ioCount++ 같은 후위 연산자의 경우 원본 값을 임시로 복사하는 메모리 할당과정이 존재
         * 반대로 전위 연산자는 이러한 과정이 없음. 하지만 이는 V8 엔진이 단독 값이면 알아서 최적화해줌
         */
        ++this.ioCount;
        callback();
      });
    } else {
      callback();
    }
  }

  /**
   * 스트림에서 end 메서드를 호출하면 이는 더 이상 보낼 데이터가 없다는 의미임
   * 해당 시점에 공식적으로 마지막 순간에 호출되는 메서드임
   */
  _final(callback: StreamCallback): void {
    if (this.bufferedBytes > 0) {
      if (this.fileDescriptor === null) {
        return callback(new Error("파일 디스크립터가 아직 준비되지 않았습니다"));
      }

      /**
       * 미처 쓰이지 못한 데이터를 마저 디스크에 기록하고 버퍼를 비움
       */
      write(this.fileDescriptor, Buffer.concat(this.ramBuffer), (error) => {
        if (error) {
          return callback(error);
        }

        ++this.ioCount;
        this.ramBuffer = [];
        this.bufferedBytes = 0;
        callback();
      });
    } else {
      callback();
    }
  }

  /**
   * 모든 파일 쓰기 작업이 정상적으로 끝났거나 중간에 에러가 발생해서 파이프가 깨지면 실행됨
   */
  _destroy(error: Error | null, callback: StreamCallback): void {
    console.log("=== 최종 결과 ===");
    console.log(`- 물리적 디스크 IO : ${this.ioCount} 회`);

    if (this.fileDescriptor !== null) {
      /**
       * OS로 부터 발급받은 파일 디스크립터를 다시 반납함
       * 해당 과정을 거치지 않으면 계속해서 점유하므로 결국 고갈이 발생할 수 있음
       */
      close(this.fileDescriptor, (closeError) => {
        callback(closeError || error);
      });
    } else {
      callback(error);
    }
  }
}

const stream = new SmartDiskWriter({
  highWaterMark: 1024 * 64,
  destination: "output.txt",
});

let currentIndex = 0;
const MAX_WRITE_COUNT = 1_000_000;
let backpressureCount = 0;

const executeWriting = () => {
  while (currentIndex < MAX_WRITE_COUNT) {
    const chunk = Buffer.from(` ${currentIndex} `, "utf-8");

    if (currentIndex === MAX_WRITE_COUNT - 1) {
      return stream.end(chunk);
    }

    const canKeepWriting = stream.write(chunk);

    if (!canKeepWriting) {
      break;
    }

    ++currentIndex;
  }
};

console.time("TIME");

executeWriting();

stream.on("drain", () => {
  ++backpressureCount;
  executeWriting();
});

stream.on("finish", () => {
  console.log(` Drain 방어 횟수 : ${backpressureCount}회`);
});

stream.on("close", () => {
  console.timeEnd("TIME");
});

// Drain 방어 횟수 : 118회
// === 최종 결과 ===
// - 물리적 디스크 IO : 121 회
// TIME: 106.617ms
```

## 커스텀 읽기 스트림

```ts
import { close, open, read } from "node:fs";
import Stream, { Readable } from "node:stream";

interface SmartDiskReaderOptions {
  highWaterMark: number;
  sourceFileName: string;
}

export class SmartDiskReader extends Readable {
  private readonly sourceFileName: string;
  private fd: number | null;

  constructor({ highWaterMark, sourceFileName }: SmartDiskReaderOptions) {
    super({ highWaterMark });

    this.sourceFileName = sourceFileName;
    this.fd = null;
  }

  /**
   * Node.js에서 자동으로 호출하는 초기화 메서드
   * 파일 오픈 이후에 클래스 자체에 fd 저장
   */
  _construct(callback: (error?: Error | null) => void): void {
    open(this.sourceFileName, "r", (error, fd) => {
      if (error) {
        callback(error);
      } else {
        this.fd = fd;
        callback();
      }
    });
  }

  _read(size: number): void {
    /**
     * 램에 데이터 저장을 위한 공간 할당
     * Buffer.alloc은 zero-fill을 수행하는데 여기선 해당 작업을 처리하지 않음
     * 극한의 성능을 위한 작업으로 일단 만들고 내부는 모두 디스크에서 읽은 데이터로 채움
     * 이후에 subarray를 통해서 불필요한 데이터 전부 잘라냄
     */
    const tempBuffer = Buffer.allocUnsafe(size);

    if (this.fd) {
      /**
       * 위에서 생성한 임시 버퍼에다가 데이터를 처음부터 끝까지 담아달라고 OS에 요청함
       * read 함수 5번째 인자(position)에 null을 전달했는데 이유는 아래와 같음
       * OS 커널이 자체적으로 관리하는 파일 오프셋 포인터를 자동으로 전진시켜서 루프를 돌아감
       * 이후에 순차적으로 데이터를 읽어와서 실제 데이터를 버퍼에 담음
       */
      read(this.fd, tempBuffer, 0, size, null, (error, bytesRead) => {
        if (error) {
          return this.destroy(error);
        }

        /**
         * 실제 읽어온 데이터가 실제로 존재하는지 꼭 검사해야함
         * else 블록의 경우 EOF로 판별함
         */
        if (bytesRead > 0) {
          /**
           * 남은 빈 공간을 잘라낼 때 메모리를 복제하는게 아닌 가벼운 뷰만 생성함
           * Buffer.allocUnsafe와 결합되어 퍼포먼스 최적화의 끝판왕임
           */
          const actualData = tempBuffer.subarray(0, bytesRead);
          this.push(actualData);
        } else {
          this.push(null);
        }
      });
    }
  }

  /**
   * OS에 fd 반납
   */
  _destroy(error: Error | null, callback: (error?: Error | null) => void): void {
    if (this.fd !== null) {
      close(this.fd, (closeError) => {
        callback(closeError || error);
      });
    } else {
      callback(error);
    }
  }
}

console.time("Custom Stream Read Time");
const stream = new SmartDiskReader({
  highWaterMark: 1024 * 64,
  sourceFileName: "high_performance_output.txt",
});

let totalChunks = 0;
let totalBytes = 0;

stream.on("data", (chunk) => {
  ++totalChunks;
  totalBytes += chunk.length;
});

stream.on("end", () => {
  // 데이터 읽기 완료
  // - CHUNK 쪼개기 횟수 : 16384회
  // - 읽은 용량 : 1024.00 MB
  // Custom Stream Read Time: 197.68ms
  console.log("데이터 읽기 완료");
  console.log(`- CHUNK 쪼개기 횟수 : ${totalChunks}회`);
  console.log(`- 읽은 용량 : ${(totalBytes / 1024 / 1024).toFixed(2)} MB`);
  console.timeEnd("Custom Stream Read Time");
});
```
